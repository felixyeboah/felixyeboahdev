'use client';

import { useCallback, useEffect, useLayoutEffect, useMemo, useRef, useState } from 'react';
import { flushSync } from 'react-dom';

import { type Photo, type PhotoSet, byColour } from '@/lib/site/photos';

import { Lightbox } from './lightbox';
import { Canvas, Reel, Wall } from './views';

/* Photographs: one archive, three ways to look at it (wall, reel, canvas), a set filter, a recent/colour
   sort and a lightbox. All of it lives in the URL (?set=streets&sort=colour&view=reel&p=<id>): filters with
   replaceState like the work index, the lightbox with pushState so Back closes it and a link opens it. */

export type View = 'wall' | 'reel' | 'canvas';
export type Sort = 'recent' | 'colour';
type State = { set: string; sort: Sort; view: View };

export type ViewProps = {
    list: Photo[];
    /** frame number of a photo in the full archive (1-based, stable across filters) */
    no: Map<string, number>;
    setTitle: Record<string, string>;
    onOpen: (id: string, el: HTMLElement | null) => void;
};

const VIEWS: Array<[View, string]> = [
    ['wall', 'Wall'],
    ['reel', 'Reel'],
    ['canvas', 'Canvas'],
];
const SORTS: Array<[Sort, string]> = [
    ['recent', 'Recent'],
    ['colour', 'Colour'],
];

const ICONS: Record<View, React.ReactNode> = {
    wall: (
        <svg viewBox="0 0 16 16" fill="none" aria-hidden="true">
            <rect x="2.5" y="2.5" width="4.5" height="6.5" rx="1" stroke="currentColor" strokeWidth="1.3" />
            <rect x="9" y="2.5" width="4.5" height="3.5" rx="1" stroke="currentColor" strokeWidth="1.3" />
            <rect x="2.5" y="11" width="4.5" height="2.5" rx="1" stroke="currentColor" strokeWidth="1.3" />
            <rect x="9" y="8" width="4.5" height="5.5" rx="1" stroke="currentColor" strokeWidth="1.3" />
        </svg>
    ),
    reel: (
        <svg viewBox="0 0 16 16" fill="none" aria-hidden="true">
            <rect x="5" y="3.5" width="6" height="9" rx="1" stroke="currentColor" strokeWidth="1.3" />
            <path d="M2 5v6M14 5v6" stroke="currentColor" strokeWidth="1.3" strokeLinecap="round" />
        </svg>
    ),
    canvas: (
        <svg viewBox="0 0 16 16" fill="none" aria-hidden="true">
            <rect x="2" y="3" width="4" height="5" rx="1" stroke="currentColor" strokeWidth="1.3" />
            <rect x="9.5" y="2" width="4.5" height="3.5" rx="1" stroke="currentColor" strokeWidth="1.3" />
            <rect x="7" y="9" width="5" height="5" rx="1" stroke="currentColor" strokeWidth="1.3" />
        </svg>
    ),
};

const motionOn = () => document.documentElement.classList.contains('motion');

type VT = { finished: Promise<void>; ready: Promise<void> };
type VTDocument = Document & { startViewTransition?: (cb: () => void) => VT };

/**
 * Runs a DOM update inside a view transition. `from` is morphed into whatever `to()` returns after the
 * update (both carry the shared name only for the length of the transition). Without the API, or with
 * reduced motion, the update just runs.
 */
function transition(update: () => void, from?: HTMLElement | null, to?: () => HTMLElement | null) {
    const doc = document as VTDocument;
    if (!doc.startViewTransition || !motionOn() || document.hidden) {
        update();
        return;
    }
    if (from) from.style.viewTransitionName = 'ph-active';
    const target: { el: HTMLElement | null } = { el: null };
    const t = doc.startViewTransition(() => {
        if (from) from.style.viewTransitionName = '';
        update();
        target.el = to?.() ?? null;
        if (target.el) target.el.style.viewTransitionName = 'ph-active';
    });
    /* a skipped transition (tab hidden mid-way, another one starting) rejects `ready`; the update has still run */
    t.ready.catch(() => {});
    t.finished
        .catch(() => {})
        .then(() => {
            if (target.el) target.el.style.viewTransitionName = '';
        });
}

const inViewport = (el: Element) => {
    const r = el.getBoundingClientRect();
    return r.bottom > 0 && r.top < innerHeight && r.right > 0 && r.left < innerWidth;
};

export function Gallery({ photos, sets }: { photos: Photo[]; sets: PhotoSet[] }) {
    const root = useRef<HTMLElement>(null);
    const [state, setState] = useState<State>({ set: 'all', sort: 'recent', view: 'wall' });
    const [open, setOpen] = useState<string | null>(null);
    const [status, setStatus] = useState('');
    const stateRef = useRef(state);
    const openRef = useRef(open);
    const pushed = useRef(false);
    const closing = useRef(false); /* history.back() is async: a second close before popstate must not go back twice */
    stateRef.current = state;
    openRef.current = open;

    const no = useMemo(() => new Map(photos.map((p, i) => [p.id, i + 1])), [photos]);
    const byId = useMemo(() => new Map(photos.map((p) => [p.id, p])), [photos]);
    const setTitle = useMemo(() => Object.fromEntries(sets.map((s) => [s.id, s.title])), [sets]);
    const list = useMemo(() => {
        const l = photos.filter((p) => state.set === 'all' || p.set === state.set);
        return state.sort === 'colour' ? [...l].sort(byColour) : l;
    }, [photos, state.set, state.sort]);

    const href = useCallback((s: State, p: string | null) => {
        const q = new URLSearchParams();
        if (s.set !== 'all') q.set('set', s.set);
        if (s.sort !== 'recent') q.set('sort', s.sort);
        if (s.view !== 'wall') q.set('view', s.view);
        if (p) q.set('p', p);
        const str = q.toString();
        return location.pathname + (str ? '?' + str : '');
    }, []);

    /* read the URL once, before paint */
    useLayoutEffect(() => {
        try {
            const q = new URLSearchParams(location.search);
            const set = q.get('set');
            const view = q.get('view');
            const next: State = {
                set: set && sets.some((s) => s.id === set) ? set : 'all',
                sort: q.get('sort') === 'colour' ? 'colour' : 'recent',
                view: view === 'reel' || view === 'canvas' ? view : 'wall',
            };
            if (next.set !== 'all' || next.sort !== 'recent' || next.view !== 'wall') setState(next);
            const p = q.get('p');
            if (p && byId.has(p)) setOpen(p);
        } catch {}
    }, [sets, byId]);

    /** the on-page frame for a photo in the current view, brought into view if it can be */
    const thumbFor = useCallback((id: string) => {
        const el = root.current?.querySelector<HTMLElement>(`[data-view] [data-id="${id}"] .fr`);
        if (!el) return null;
        if (!inViewport(el) && !el.closest('.cv')) el.scrollIntoView({ block: 'center', inline: 'center', behavior: 'instant' });
        return inViewport(el) ? el : null;
    }, []);

    const shut = useCallback(() => {
        const id = openRef.current;
        if (!id) return;
        transition(
            () => flushSync(() => setOpen(null)),
            document.querySelector<HTMLElement>('.lb__frame'),
            () => thumbFor(id),
        );
    }, [thumbFor]);

    const onOpen = useCallback(
        (id: string, el: HTMLElement | null) => {
            history.pushState(null, '', href(stateRef.current, id));
            pushed.current = true;
            transition(
                () => flushSync(() => setOpen(id)),
                el,
                () => document.querySelector<HTMLElement>('.lb__frame'),
            );
        },
        [href],
    );

    const onStep = useCallback(
        (id: string) => {
            setOpen(id);
            history.replaceState(history.state, '', href(stateRef.current, id));
        },
        [href],
    );

    const onClose = useCallback(() => {
        if (closing.current) return;
        if (pushed.current) {
            closing.current = true;
            history.back(); /* popstate does the closing */
            return;
        }
        history.replaceState(history.state, '', href(stateRef.current, null));
        shut();
    }, [href, shut]);

    useEffect(() => {
        const onPop = () => {
            const p = new URLSearchParams(location.search).get('p');
            pushed.current = false;
            closing.current = false;
            if (p && byId.has(p)) {
                pushed.current = true;
                setOpen(p);
            } else shut();
        };
        window.addEventListener('popstate', onPop);
        return () => window.removeEventListener('popstate', onPop);
    }, [byId, shut]);

    const choose = (next: Partial<State>) => {
        const s = { ...state, ...next };
        if (s.set === state.set && s.sort === state.sort && s.view === state.view) return;
        const n = photos.filter((p) => s.set === 'all' || p.set === s.set).length;
        history.replaceState(history.state, '', href(s, null));
        transition(() =>
            flushSync(() => {
                setState(s);
                setStatus(`${s.set === 'all' ? 'All sets' : setTitle[s.set]}, ${n} ${n === 1 ? 'photograph' : 'photographs'}, ${s.view} view, sorted by ${s.sort}`);
            }),
        );
    };

    const viewProps: ViewProps = { list, no, setTitle, onOpen };

    return (
        <section className="pg" aria-labelledby="pg-title" ref={root}>
            <h2 className="sr-only" id="pg-title">
                All photographs
            </h2>
            <div className="pbar">
                <div className="wrap pbar__in">
                    <div className="fchips" role="group" aria-label="Filter by set">
                        <button className="fchip" type="button" aria-pressed={state.set === 'all'} onClick={() => choose({ set: 'all' })}>
                            All <span>{photos.length}</span>
                        </button>
                        {sets.map((s) => (
                            <button key={s.id} className="fchip" type="button" aria-pressed={state.set === s.id} onClick={() => choose({ set: s.id })}>
                                {s.title} <span>{s.count}</span>
                            </button>
                        ))}
                    </div>
                    <div className="pbar__segs">
                        <div className="seg" role="group" aria-label="Sort">
                            {SORTS.map(([k, label]) => (
                                <button key={k} type="button" aria-pressed={state.sort === k} onClick={() => choose({ sort: k })}>
                                    {label}
                                </button>
                            ))}
                        </div>
                        <div className="seg" role="group" aria-label="View">
                            {VIEWS.map(([k, label]) => (
                                <button key={k} type="button" aria-pressed={state.view === k} aria-label={label} onClick={() => choose({ view: k })}>
                                    {ICONS[k]}
                                    <span>{label}</span>
                                </button>
                            ))}
                        </div>
                    </div>
                </div>
                <p className="sr-only" role="status" aria-live="polite">
                    {status}
                </p>
            </div>

            <div data-view={state.view}>
                {state.view === 'wall' && <Wall {...viewProps} />}
                {state.view === 'reel' && <Reel {...viewProps} />}
                {state.view === 'canvas' && <Canvas {...viewProps} />}
            </div>

            <Cursor scope={root} />

            {open && byId.has(open) && (
                <Lightbox
                    list={list.some((p) => p.id === open) ? list : photos}
                    id={open}
                    no={no}
                    setTitle={setTitle}
                    onStep={onStep}
                    onClose={onClose}
                />
            )}
        </section>
    );
}

/** A label that trails the pointer over anything marked data-cur="View" (fine pointers, motion on). */
function Cursor({ scope }: { scope: React.RefObject<HTMLElement | null> }) {
    const el = useRef<HTMLDivElement>(null);

    useEffect(() => {
        const host = scope.current;
        const dot = el.current;
        if (!host || !dot || !motionOn() || !window.matchMedia('(hover: hover) and (pointer: fine)').matches) return;
        host.setAttribute('data-cursor', '');
        let x = 0;
        let y = 0;
        let tx = 0;
        let ty = 0;
        let raf = 0;
        let on = false;
        const tick = () => {
            x += (tx - x) * 0.22;
            y += (ty - y) * 0.22;
            dot.style.transform = `translate3d(${x}px,${y}px,0)`;
            raf = on || Math.abs(tx - x) > 0.5 || Math.abs(ty - y) > 0.5 ? requestAnimationFrame(tick) : 0;
        };
        const move = (e: PointerEvent) => {
            if (e.pointerType !== 'mouse') return;
            tx = e.clientX;
            ty = e.clientY;
            const target = (e.target as Element | null)?.closest?.('[data-cur]');
            const next = !!target;
            if (next) dot.textContent = target!.getAttribute('data-cur');
            if (next && !on) {
                x = tx;
                y = ty;
            }
            on = next;
            dot.classList.toggle('is-on', on);
            if (!raf) raf = requestAnimationFrame(tick);
        };
        const leave = () => {
            on = false;
            dot.classList.remove('is-on');
        };
        const down = () => dot.classList.add('is-down');
        const up = () => dot.classList.remove('is-down');
        host.addEventListener('pointermove', move);
        host.addEventListener('pointerleave', leave);
        host.addEventListener('pointerdown', down);
        window.addEventListener('pointerup', up);
        window.addEventListener('scroll', leave, { passive: true });
        return () => {
            cancelAnimationFrame(raf);
            host.removeAttribute('data-cursor');
            host.removeEventListener('pointermove', move);
            host.removeEventListener('pointerleave', leave);
            host.removeEventListener('pointerdown', down);
            window.removeEventListener('pointerup', up);
            window.removeEventListener('scroll', leave);
        };
    }, [scope]);

    return <div className="pcur mono" ref={el} aria-hidden="true" />;
}
