import type { Metadata } from 'next';
import Link from 'next/link';
import type { CSSProperties } from 'react';

import { siteConfig } from '@/config/site';
import { PageStyle } from '@/core/site/chrome';
import { HAS_SAMPLES, PHOTOS, SETS, pad3, photoFull } from '@/lib/site/photos';

import { Gallery } from './gallery';
import { HeroTrail } from './hero';

/* Photographs. Data comes from the manifest behind lib/site/photos.ts (see that file for how to add pictures). */

export const metadata: Metadata = {
    title: { absolute: 'Photos — Felix Yeboah' },
    description: 'Photographs by Felix Yeboah.',
    alternates: { canonical: '/photos' },
    /* placeholder stock is on the page until the real photographs replace it; keep that out of search */
    ...(HAS_SAMPLES ? { robots: { index: false, follow: true } } : {}),
};

const TITLE = 'Photographs';

export default function PhotosPage() {
    const own = PHOTOS.filter((p) => !p.sample);
    const years = PHOTOS.map((p) => p.date?.slice(0, 4)).filter(Boolean) as string[];
    const span = years.length ? [years[years.length - 1], years[0]].filter((y, i, a) => a.indexOf(y) === i).join('–') : null;
    /* a spread of the archive for the hero trail, not just the newest dozen */
    const stride = Math.max(1, Math.floor(PHOTOS.length / 12));
    const trail = PHOTOS.filter((_, i) => i % stride === 0).slice(0, 12);

    const jsonLd = own.length
        ? {
              '@context': 'https://schema.org',
              '@type': 'ImageGallery',
              name: 'Photographs by Felix Yeboah',
              url: siteConfig.url + '/photos',
              author: { '@type': 'Person', name: siteConfig.author, url: siteConfig.url },
              image: own.slice(0, 24).map((p) => ({
                  '@type': 'ImageObject',
                  contentUrl: siteConfig.url + photoFull(p),
                  width: p.w,
                  height: p.h,
                  creator: { '@type': 'Person', name: siteConfig.author },
                  ...(p.title ? { name: p.title } : {}),
                  ...(p.location ? { contentLocation: p.location } : {}),
              })),
          }
        : null;

    return (
        <main id="main" data-fy-skip="peek">
            <PageStyle name="photos" />
            <header className="phero" data-instant="">
                <HeroTrail photos={trail} />
                <div className="wrap phero__in">
                    <p className="pagehead__crumb mono" data-reveal="">
                        <Link href="/">Index</Link>
                        <span className="sep" aria-hidden="true" />
                        <span aria-current="page">Photos</span>
                    </p>
                    <h1 className="phero__title" aria-label={TITLE}>
                        {TITLE.split('').map((ch, i) => (
                            <span key={i} aria-hidden="true" style={{ '--i': i } as CSSProperties}>
                                {ch}
                            </span>
                        ))}
                    </h1>
                    <dl className="phero__meta mono" data-reveal="" style={{ '--d': '520ms' } as CSSProperties}>
                        <div>
                            <dt>Frames</dt>
                            <dd>{pad3(PHOTOS.length)}</dd>
                        </div>
                        <div>
                            <dt>Sets</dt>
                            <dd>{pad3(SETS.length)}</dd>
                        </div>
                        {span && (
                            <div>
                                <dt>Years</dt>
                                <dd>{span}</dd>
                            </div>
                        )}
                    </dl>
                </div>
            </header>
            <Gallery photos={PHOTOS} sets={SETS} />
            {jsonLd && <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd).replace(/</g, '\\u003c') }} />}
        </main>
    );
}
