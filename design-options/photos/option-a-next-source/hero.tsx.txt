'use client';

import { useEffect, useRef } from 'react';

import { type Photo, photoSrc } from '@/lib/site/photos';

/* The pointer leaves a trail of photographs behind the title. Left alone (or on touch screens) the trail
   draws itself along a slow figure-eight. Decoration only: nothing here is reachable or announced, and
   with reduced motion it never starts. */

const POOL = 9;
const GAP = 96; /* px of travel between two photographs */
const LIFE = 1500;

export function HeroTrail({ photos }: { photos: Photo[] }) {
    const ref = useRef<HTMLDivElement>(null);

    useEffect(() => {
        const layer = ref.current;
        const hero = layer?.parentElement;
        if (!layer || !hero || !photos.length || !document.documentElement.classList.contains('motion')) return;

        const imgs = Array.from(layer.querySelectorAll('img'));
        const ready = new Set<string>();
        photos.forEach((p) => {
            const im = new Image();
            im.onload = () => ready.add(p.id);
            im.src = photoSrc(p, p.widths[0]);
        });

        let slot = 0;
        let shot = 0;
        let z = 1;
        let lastX = -1e4;
        let lastY = -1e4;
        let lastMove = performance.now() - 1500; /* the self-drawn trail starts just after the title lands */
        let visible = true;
        let raf = 0;

        const spawn = (x: number, y: number) => {
            let p = photos[shot % photos.length];
            for (let i = 0; i < photos.length && !ready.has(p.id); i++) p = photos[++shot % photos.length];
            if (!ready.has(p.id)) return;
            shot++;
            const el = imgs[slot++ % imgs.length];
            const base = layer.clientWidth < 700 ? 150 : 250;
            const r = p.w / p.h;
            const w = r > 1 ? base * 1.25 : base;
            el.src = photoSrc(p, p.widths[0]);
            el.style.width = w + 'px';
            el.style.height = w / r + 'px';
            el.style.left = x - w / 2 + 'px';
            el.style.top = y - w / r / 2 + 'px';
            el.style.zIndex = String(z++);
            const tilt = ((shot * 37) % 9) - 4;
            el.getAnimations().forEach((a) => a.cancel());
            el.animate(
                [
                    { opacity: 0, transform: `scale(.55) rotate(${tilt * 1.6}deg)`, easing: 'cubic-bezier(.16,1,.3,1)' },
                    { opacity: 1, transform: `scale(1) rotate(${tilt}deg)`, offset: 0.16 },
                    { opacity: 1, transform: `scale(1) rotate(${tilt}deg)`, offset: 0.62, easing: 'cubic-bezier(.4,0,.2,1)' },
                    { opacity: 0, transform: `translateY(36px) scale(.94) rotate(${tilt}deg)` },
                ],
                { duration: LIFE, easing: 'linear', fill: 'both' },
            );
        };

        const at = (x: number, y: number) => {
            if (Math.hypot(x - lastX, y - lastY) < GAP) return;
            lastX = x;
            lastY = y;
            spawn(x, y);
        };

        const onMove = (e: PointerEvent) => {
            const r = layer.getBoundingClientRect();
            lastMove = performance.now();
            at(e.clientX - r.left, e.clientY - r.top);
        };

        /* self-drawing figure-eight while nobody is steering */
        const auto = (t: number) => {
            raf = 0;
            if (!visible || document.hidden) return;
            if (t - lastMove > 2200) {
                const w = layer.clientWidth;
                const h = layer.clientHeight;
                const k = t / 2600;
                at(w / 2 + Math.sin(k) * w * 0.34, h * 0.48 + Math.sin(k * 2) * h * 0.2);
            }
            raf = requestAnimationFrame(auto);
        };
        const kick = () => {
            if (!raf) raf = requestAnimationFrame(auto);
        };

        const io = new IntersectionObserver(([en]) => {
            visible = en.isIntersecting;
            if (visible) kick();
        });
        io.observe(hero);
        hero.addEventListener('pointermove', onMove);
        document.addEventListener('visibilitychange', kick);
        kick();
        return () => {
            cancelAnimationFrame(raf);
            io.disconnect();
            hero.removeEventListener('pointermove', onMove);
            document.removeEventListener('visibilitychange', kick);
            imgs.forEach((el) => el.getAnimations().forEach((a) => a.cancel()));
        };
    }, [photos]);

    return (
        <div className="phero__trail" ref={ref} aria-hidden="true">
            {Array.from({ length: POOL }, (_, i) => (
                <img key={i} alt="" decoding="async" draggable={false} />
            ))}
        </div>
    );
}
